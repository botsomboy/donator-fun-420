# Donator - Fun 420

A small cosmetic RuneLite plugin.

## Features

- **4:20 alarm** — at 16:20 and 04:20 local time a pulsing coloured border and a
  message appear over the game screen for 60 seconds. If you log in halfway
  through that minute, you still get the full 60 seconds.
- **April 20 banner** — on April 20 a green banner reading "Happy 420 today"
  sits at the top of the screen. Click the × to dismiss it for the rest of the
  day; it comes back next year.

Colours and texts are configurable, and both features can be switched off
independently. The Test section lets you trigger the alarm and preview the
banner on any day.

The plugin only draws on screen. It does not read account data, make network
requests, or interact with the game in any way.

./gradlew build
./gradlew run
```

package com.donator.fun420;

import java.time.Clock;
import java.time.Duration;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.Month;

/**
 * Levert de tijd waar de plugin op werkt en beantwoordt de twee kalendervragen.
 * De basisklok is injecteerbaar, zodat tests en de simulatieschakelaar exact
 * hetzelfde pad gebruiken als productie.
 */
public class Fun420Clock
{
	private static final int ALARM_MINUTE = 20;
	private static final int AFTERNOON_HOUR = 16;
	private static final int NIGHT_HOUR = 4;
	private static final int BANNER_DAY_OF_MONTH = 20;

	private final Clock baseClock;
	private volatile Clock clock;

	public Fun420Clock()
	{
		this(Clock.systemDefaultZone());
	}

	public Fun420Clock(Clock baseClock)
	{
		this.baseClock = baseClock;
		this.clock = baseClock;
	}

	public LocalDateTime now()
	{
		return LocalDateTime.now(clock);
	}

	public void useRealDate()
	{
		this.clock = baseClock;
	}

	public void simulateApril20()
	{
		this.clock = Clock.offset(baseClock, offsetToApril20(LocalDateTime.now(baseClock)));
	}

	static Duration offsetToApril20(LocalDateTime from)
	{
		LocalDateTime target = from
			.withMonth(Month.APRIL.getValue())
			.withDayOfMonth(BANNER_DAY_OF_MONTH);
		return Duration.between(from, target);
	}

	static boolean isAlarmWindow(LocalDateTime time)
	{
		if (time.getMinute() != ALARM_MINUTE)
		{
			return false;
		}
		return time.getHour() == AFTERNOON_HOUR || time.getHour() == NIGHT_HOUR;
	}

	static boolean isBannerDay(LocalDate date)
	{
		return date.getMonth() == Month.APRIL && date.getDayOfMonth() == BANNER_DAY_OF_MONTH;
	}
}
```

package com.donator.fun420;

import java.time.Duration;
import java.time.LocalDateTime;

/**
 * Houdt bij of het alarm loopt. Het alarm duurt altijd 60 seconden vanaf het
 * moment dat het startte, ook als dat midden in de 420-minuut was.
 */
public class AlarmState
{
	static final Duration DURATION = Duration.ofSeconds(60);
	private static final long PULSE_PERIOD_MILLIS = 1000L;

	private volatile LocalDateTime startedAt;

	/**
	 * Start het alarm als de klok het venster binnengaat terwijl de speler
	 * ingelogd is, of als de speler inlogt terwijl de klok al in het venster
	 * staat. Uitloggen wist de state, zodat opnieuw inloggen binnen hetzelfde
	 * venster een verse minuut oplevert.
	 */
	public void update(LocalDateTime now, boolean loggedIn)
	{
		if (!loggedIn)
		{
			reset();
			return;
		}

		if (Fun420Clock.isAlarmWindow(now) && !isActive(now))
		{
			start(now);
		}
	}

	public void start(LocalDateTime now)
	{
		startedAt = now;
	}

	public void reset()
	{
		startedAt = null;
	}

	public boolean isActive(LocalDateTime now)
	{
		LocalDateTime start = startedAt;
		if (start == null || now.isBefore(start))
		{
			return false;
		}
		return Duration.between(start, now).compareTo(DURATION) < 0;
	}

	/**
	 * Waarde tussen 0 en 1 die eens per seconde op en neer gaat; de overlay
	 * vertaalt hem naar een alpha.
	 */
	public double pulse(LocalDateTime now)
	{
		LocalDateTime start = startedAt;
		if (start == null || !isActive(now))
		{
			return 0.0;
		}

		long millis = Duration.between(start, now).toMillis();
		double phase = (millis % PULSE_PERIOD_MILLIS) / (double) PULSE_PERIOD_MILLIS;
		return (1.0 - Math.cos(2.0 * Math.PI * phase)) / 2.0;
	}
}

		}

		LocalDateTime now = plugin.getClock().now();
		AlarmState state = plugin.getAlarmState();
		if (!state.isActive(now))

			return null;
		}

		int width = client.getCanvasWidth();

		Color color = withPulseAlpha(config.alarmColor(), state.pulse(now));

		Stroke originalStroke = graphics.getStroke();
		graphics.setColor(color);
		graphics.setStroke(new BasicStroke(BORDER_WIDTH));
		graphics.drawRect(
			BORDER_WIDTH / 2,
			BORDER_WIDTH / 2,
			width - BORDER_WIDTH,
			height - BORDER_WIDTH);
		graphics.setStroke(originalStroke);

		Font originalFont = graphics.getFont();
		graphics.setFont(originalFont.deriveFont(Font.BOLD, TEXT_SIZE));

		String text = config.alarmText();
		FontMetrics metrics = graphics.getFontMetrics();
		int textX = (width - metrics.stringWidth(text)) / 2;
		int textY = height / 2;

		graphics.setColor(Color.BLACK);
		graphics.drawString(text, textX + 2, textY + 2);
		graphics.setColor(color);
		graphics.drawString(text, textX, textY);

		graphics.setFont(originalFont);
		return null;
	}

	private static Color withPulseAlpha(Color base, double pulse)
	{
		int alpha = (int) Math.round(MIN_ALPHA + ((MAX_ALPHA - MIN_ALPHA) * pulse));
		alpha = Math.max(0, Math.min(MAX_ALPHA, alpha));
		return new Color(base.getRed(), base.getGreen(), base.getBlue(), alpha);
	}

		bannerDismissedOn = today;

		applySimulation();
		overlayManager.add(alarmOverlay);
		overlayManager.add(bannerOverlay);

	{
		mouseManager.unregisterMouseListener(bannerMouseListener);
		overlayManager.remove(alarmOverlay);
		overlayManager.remove(bannerOverlay);
		alarmState.reset();
		clock.useRealDate();
		log.debug("Donator - Fun 420 stopped");
	}

		alarmState.update(clock.now(), client.getGameState() == GameState.LOGGED_IN);
	}

	public void onGameStateChanged(GameStateChanged event)
	{
		if (event.getGameState() != GameState.LOGGED_IN)
		{
			alarmState.reset();
		}
	}

	@Subscribe
	public void onConfigChanged(ConfigChanged event)

					Fun420Config.GROUP,
					Fun420Config.KEY_TEST_ALARM,

			}

		{
			applySimulation();
		}
	}
